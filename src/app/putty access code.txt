app.get('/api/employees/salary-clusters', async (req, res) => {
  const [rows] = await pool.query('SELECT sonbor_salary FROM Sonbor_Employees');
  const salaries = rows.map(r => Number(r.sonbor_salary)).filter(v => v > 0).sort((a, b) => a - b);

  if (salaries.length === 0) return res.json({ success: true, clusters: [] });

  let centroids = [salaries[0], salaries[Math.floor(salaries.length / 2)], salaries[salaries.length - 1]];
  let clusters = [[], [], []];

  for (let i = 0; i < 50; i++) {
    clusters = [[], [], []];
    salaries.forEach(val => {
      let idx = 0, minDist = Math.abs(val - centroids[0]);
      for (let c = 1; c < 3; c++) {
        let dist = Math.abs(val - centroids[c]);
        if (dist < minDist) { minDist = dist; idx = c; }
      }
      clusters[idx].push(val);
    });
    clusters.forEach((cluster, c) => {
      if (cluster.length > 0) centroids[c] = cluster.reduce((a, b) => a + b, 0) / cluster.length;
    });
  }

  const result = centroids.map((cen, idx) => ({
    centroid: clusters[idx].length ? Math.round(cen) : 0,
    count: clusters[idx].length
  })).sort((a, b) => a.centroid - b.centroid).map((item, idx) => ({
    group: `Tier ${idx + 1}`,
    centroid: item.centroid,
    count: item.count
  }));

  res.json({ success: true, clusters: result });
});

// AI Salary Prediction & Recommendation API
app.post('/api/employees/ai-salary-prediction', async (req, res) => {
  try {
    const { currentSalary, position } = req.body;
    const salary = Number(currentSalary);

    if (isNaN(salary) || salary <= 0) {
      return res.status(400).json({ error: 'Valid current salary is required.' });
    }

    // ดึงเงินเดือนทั้งหมดในระบบมาใช้อ้างอิงค่าเฉลี่ยตลาดภายในองค์กร
    const [rows] = await pool.query('SELECT sonbor_salary, sonbor_position FROM Sonbor_Employees');
    const allSalaries = rows.map(r => Number(r.sonbor_salary));
    const marketAvg = allSalaries.length > 0 ? allSalaries.reduce((a, b) => a + b, 0) / allSalaries.length : salary;

    // คำนวณความสมเหตุสมผล (Salary Health Check)
    let healthStatus = 'Fair & Competitive';
    let suggestedIncreasePercent = 7.5; // ค่าเริ่มต้นมาตรฐาน (Midpoint Merit)
    let aiReasoning = '';

    if (salary < marketAvg * 0.7) {
      healthStatus = 'Underpaid (Risk of Turnover)';
      suggestedIncreasePercent = 12.0; // ควรปรับขึ้นสูงหน่อย เพื่อให้พนักงานพอใจและพ้นเกณฑ์เสี่ยงลาออก
      aiReasoning = 'AI Analysis: Employee salary is significantly below the internal market average. A higher merit increase of 12% is strongly recommended to boost retention and fairness.';
    } else if (salary > marketAvg * 1.4) {
      healthStatus = 'Overpaid / Senior Level';
      suggestedIncreasePercent = 4.0; // ปรับขึ้นในอัตราที่น้อยลง เพราะฐานเงินเดือนสูงอยู่แล้ว
      aiReasoning = 'AI Analysis: Salary is in the upper tier. A modest merit increase of 4% is recommended to maintain cost efficiency while rewarding performance.';
    } else {
      healthStatus = 'Balanced & Reasonable';
      suggestedIncreasePercent = 8.0; // อัตราปกติที่สมเหตุสมผล
      aiReasoning = 'AI Analysis: Salary aligns well with internal benchmarks. An 8% increase provides optimal satisfaction and matches performance expectations.';
    }

    const recommendedNewSalary = Math.round(salary * (1 + suggestedIncreasePercent / 100));
    const monthlyDifference = recommendedNewSalary - salary;

    return res.status(200).json({
      success: true,
      prediction: {
        currentSalary: salary,
        marketAverage: Math.round(marketAvg),
        healthStatus,
        suggestedIncreasePercent,
        recommendedNewSalary,
        monthlyDifference,
        aiReasoning
      }
    });
  } catch (error) {
    return res.status(500).json({ success: false, error: 'AI prediction failed' });
  }
});
